1 querySelectorAll + forEach — 메뉴 카드 클릭

BREWVENTI 탭 버튼에 이벤트를 등록할 때 querySelectorAll + forEach를 사용합니다.
카드를 클릭하면 "전체 초기화 → 하나만 활성화" 패턴이 동작합니다.

// querySelectorAll: 조건에 맞는 요소 전부 선택 → NodeList 반환
const cards = document.querySelectorAll('.menu-card');

cards.forEach(card => {
  card.addEventListener('click', () => {
    // 전체 초기화
    cards.forEach(c => c.classList.remove('selected'));
    // 클릭된 것만 활성화
    card.classList.add('selected');
  });
});

2 setInterval — 카운트다운 타이머

BREWVENTI 슬라이더는 setInterval로 3초마다 자동 전환됩니다.
타이머를 시작/정지하면서 setInterval과 clearInterval 사용법을 확인하세요.

10
let count = 10;
let timerId = null; // ID를 저장해야 정지 가능!

// 시작: 1초마다 반복
timerId = setInterval(() => {
  count--;
  // 화면 업데이트
}, 1000);

// 정지
clearInterval(timerId);

3 data-* 속성 — 탭 카테고리 읽기

BREWVENTI 탭 버튼은 data-category 속성으로 카테고리 값을 저장합니다.
탭을 클릭하면 dataset으로 값을 읽어서 어떤 카테고리인지 표시합니다.

탭을 클릭하면 data-* 속성 값을 읽어옵니다.
// data-category="coffee" → dataset.category → "coffee"
// data-label="커피류" → dataset.label → "커피류"

let current = 0;
const slides = document.querySelectorAll('.mini-slide');
const dots = document.querySelectorAll('.mini-dot');

function goTo(index) {
  slides[current].classList.remove('active');
  dots[current].classList.remove('active');
  current = index;
  slides[current].classList.add('active');
  dots[current].classList.add('active');
}

// 자동 전환: 나머지 연산(%)으로 0→1→2→0 순환
setInterval(() => goTo((current + 1) % slides.length), 3000);
